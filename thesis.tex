\section{First Principles: Boolean Lattice and Branchless Algebra}
\label{sec:boolean}
%% ════════════════════════════════════════════════════════════════════════════

We build the entire scheduling calculus on two foundations: the Boolean lattice
$\Lattice{\B^n,\subseteq}$ and the integer ring $\Z/2^{64}\Z$.
No further axioms are assumed.

\subsection{Boolean Lattice}

\begin{definition}[Boolean lattice $\B^n$]
Let $\B = \{0,1\}$ and $n \in \N$.
The set $\B^n$ of $n$-bit words forms a \emph{complete Boolean lattice}
$\Lattice{\B^n, \subseteq, \cap, \cup, \neg, \bot, \top}$ where:
\begin{align*}
  a \subseteq b &\iff a \;\&\; b = a           &&\text{(bitwise AND = subset)}\\
  a \cap b      &= a \;\&\; b                  &&\text{(meet = AND)}\\
  a \cup b      &= a \;|\; b                   &&\text{(join = OR)}\\
  \neg a        &= \;\sim\! a = \texttt{!a}    &&\text{(complement = NOT)}\\
  \bot          &= 0^n                          &&\text{(bottom = zero word)}\\
  \top          &= 1^n = \texttt{!0u64}         &&\text{(top = all-ones word)}
\end{align*}
\end{definition}

The partial order $\subseteq$ is the \emph{subset} ordering on bit positions.
Every subset of $\B^n$ has a meet and join, making it a complete lattice.
This is the only structure we need to reason about op masks.

\subsection{The Wrapping-Negation Identity}

The core branchless primitive in bcinr-powl is the wrapping-negation:

\begin{lemma}[Wrapping-negation selection]
\label{lem:wrapmask}
In $\Z/2^{64}\Z$, for any $b \in \{0,1\}$:
\[
  \mathtt{wrapping\_neg}(b) =
  \begin{cases}
    \top = \texttt{!0u64} & \text{if } b = 1 \\
    \bot = 0              & \text{if } b = 0
  \end{cases}
\]
\end{lemma}
\begin{proof}
In two's complement: $-1 \bmod 2^{64} = 2^{64}-1 = \texttt{0xFFFF\_FFFF\_FFFF\_FFFF} = \top$.
$-0 \bmod 2^{64} = 0 = \bot$. \qed
\end{proof}

\subsection{The Pred-Satisfaction Function}

\begin{definition}[Predicate satisfaction]
\label{def:pred_sat}
For $\done, \pred \in \B^{64}$, define:
\[
  \mathrm{sat}(\done, \pred) \;=\; \overline{(\pred \cap \overline{\done}) \neq 0}
  \;=\; \begin{cases} \top & \text{if } \pred \subseteq \done \\ \bot & \text{otherwise} \end{cases}
\]
\end{definition}

\begin{theorem}[Correctness of \texttt{pred\_satisfied}]
\label{thm:pred_sat}
The Rust expression
\lstinline|0u64.wrapping_sub((required & !done == 0) as u64)|
computes $\mathrm{sat}(\done, \pred)$ exactly.
\end{theorem}
\begin{proof}
Let $u = \pred \cap \overline{\done}$.
$u = 0 \iff \pred \subseteq \done$ (subset characterisation in $\B^{64}$).
Let $b = (u = 0)\;\text{as}\;\texttt{u64} \in \{0,1\}$.
By Lemma~\ref{lem:wrapmask}, $\texttt{wrapping\_neg}(b) = \top$ if $b=1$, $\bot$ if $b=0$,
which equals $\mathrm{sat}(\done,\pred)$. \qed
\end{proof}

\begin{remark}
The implementation uses \texttt{0u64.wrapping\_sub(b)} rather than \texttt{b.wrapping\_neg()}
because the ARM/x86 calling conventions may emit a separate NEG instruction for the latter;
the subtraction form fuses with adjacent operations in the pipeline.
On Apple M-series the three-instruction sequence \texttt{AND / SUBS / CSINV} or
\texttt{AND / TST / CSETM} produces the full mask in $\sim$1.5 cycles.
\end{remark}

\section{Fixed-Point Theory: The Chatman Equation\texorpdfstring{}{: mu(O*)}}
\label{sec:chatman}
%% ════════════════════════════════════════════════════════════════════════════

\subsection{Complete Partial Orders and Monotone Operators}

\begin{definition}[Complete partial order (CPO)]
A set $D$ with relation $\sqsubseteq$ is a \emph{CPO} if:
(a) $\sqsubseteq$ is a partial order, and
(b) every $\omega$-chain $d_0 \sqsubseteq d_1 \sqsubseteq \cdots$ has a least upper bound
$\bigsqcup_i d_i \in D$.
\end{definition}

\begin{definition}[Monotone and continuous operators]
An operator $F: D \to D$ is:
\begin{align*}
  \text{\emph{monotone}}   &\iff x \sqsubseteq y \implies F(x) \sqsubseteq F(y) \\
  \text{\emph{continuous}} &\iff F\!\left(\bigsqcup_i d_i\right) = \bigsqcup_i F(d_i)
\end{align*}
\end{definition}

\begin{theorem}[Knaster--Tarski, 1955]
\label{thm:kt}
Let $\Lattice{L,\sqsubseteq}$ be a complete lattice and $F: L \to L$ monotone.
Then $F$ has a least fixed point:
\[
  \mu(F) = \bigcap\{x \in L \mid F(x) \sqsubseteq x\}
\]
and $\mu(F) = F(\mu(F))$.
\end{theorem}

\subsection{The Causal Receipt Lattice}

\begin{definition}[Causal receipt chain]
A \emph{causal receipt} is a finite sequence of \texttt{OcelCausalFrame}s
$\sigma = (f_1, f_2, \ldots, f_k)$ where each frame $f_i$ encodes:
\begin{itemize}
  \item An instruction identity $\mathrm{id}_i \in \N$ (monotone)
  \item A denial polarity $\delta_i \in \B^{64}$ (8 verdict lanes)
  \item A BLAKE3 hash $h_i = \mathrm{BLAKE3}(h_{i-1} \,\|\, \mathrm{bytes}(f_i))$
  \item Object references $O_i \subseteq \mathrm{Obj}$
\end{itemize}
\end{definition}

\begin{definition}[Receipt prefix order]
For receipts $\sigma, \tau$, define $\sigma \sqsubseteq \tau$ iff
$\sigma$ is a prefix of $\tau$ (every frame of $\sigma$ appears in $\tau$ in order
with the same BLAKE3 chain).
The set of all finite receipts under $\sqsubseteq$ forms a CPO
$\Lattice{\mathrm{Receipts}, \sqsubseteq}$ with bottom element $\varepsilon$
(the genesis receipt).
\end{definition}

\subsection{The $\Ostar$ Operator}

\begin{definition}[The $\Ostar$ operator]
$\Ostar : \mathrm{Receipts} \to \mathrm{Receipts}$ maps a receipt $\sigma$ to the
receipt obtained by:
\begin{enumerate}
  \item Admitting $\sigma$ through the POWL scheduler (one $\mu$-step)
  \item Emitting a new \texttt{OcelCausalFrame} encoding the result
  \item Extending $\sigma$ with the new frame
  \item Verifying the extended receipt against the ConformancePredicate
\end{enumerate}
\end{definition}

\begin{proposition}[$\Ostar$ is monotone]
$\sigma \sqsubseteq \tau \implies \Ostar(\sigma) \sqsubseteq \Ostar(\tau)$.
\end{proposition}
\begin{proof}
The BLAKE3 chain is deterministic: given the same prefix, the same extension is produced.
The scheduler's done-mask is monotone (Theorem~\ref{thm:monotone_done} below).
Therefore a longer receipt $\tau$ produces a longer extension, and $\sqsubseteq$ is preserved.
\qed
\end{proof}

\begin{definition}[The Chatman Equation]
\[
  \boxed{\Reality \vdash \Agency = \mu(\Ostar)}
\]
where $\Reality$ is the initial receipt (genesis BLAKE3 hash over 32 zero bytes),
and $\mu(\Ostar)$ is the least fixed point of $\Ostar$ from Theorem~\ref{thm:kt}.
\end{definition}

\subsection{POWL Nodes and Traces}

\begin{definition}[POWL node (Kourani \& van der Aalst 2026)]
A POWL node $N$ is defined inductively:
\begin{align*}
  N &::= a \;\mid\; \tau \;\mid\; (G,E) \;\mid\; X(G) \;\mid\; \mathrm{Loop}(b,r)
\end{align*}
where $a \in \Sigma$ is an activity label, $\tau$ is the silent transition,
$(G,E)$ is a partial order with children $G$ and strict order $E \subseteq G \times G$,
$X(G)$ is an exclusive choice, and $\mathrm{Loop}(b,r)$ has body $b$ and redo $r$.
\end{definition}

\begin{definition}[Shuffle $\uplus_\rho$]
For a partial order $(G,E)$, the \emph{shuffle} $\uplus_\rho(G,E)$ is:
\[
  L(G,E) = \left\{ w_1 \cdot w_2 \cdots w_{|G|}
  \;\middle|\;
  \begin{array}{l}
    w_i \in L(g_i),\;g_i \in G,\\
    \text{and } g_i \to g_j \in E \implies
    w_i \text{ precedes } w_j \text{ in } w_1\cdots w_{|G|}
  \end{array}
  \right\}
\]
The shuffle is the set of all interleavings of child traces respecting $E$.
\end{definition}

\section{Monotone Token Calculus}
\label{sec:petri}
%% ════════════════════════════════════════════════════════════════════════════

\subsection{Token Marking as a Lattice Element}

\begin{definition}[Token marking]
A \emph{marking} $M \in \B^{64}$ assigns a token ($1$) or no token ($0$) to each
op-slot. The set of markings forms the Boolean lattice $\Lattice{\B^{64},\subseteq}$
from Section~\ref{sec:boolean}.
\end{definition}

\begin{definition}[Monotone done-set]
A sequence of markings $(M_0, M_1, M_2, \ldots)$ is \emph{monotone} iff
$M_t \subseteq M_{t+1}$ for all $t$.
\end{definition}

The key invariant of bcinr-powl is that the \texttt{done} mask is always monotone:
once an op fires, it is done forever.

\subsection{Standard vs Monotone Petri Net Firing}

A standard Petri net transition $T = (\mathrm{input}, \mathrm{output})$ fires on
marking $M$ by:
\[
  M' = (M \setminus \mathrm{input}) \cup \mathrm{output}
\]
This is \emph{destructive}: tokens in $\mathrm{input}$ are consumed.
The done-set would not be monotone under standard firing.

\begin{definition}[Monotone Petri mapping]
\label{def:monotone_map}
For POWL op $i$ with predecessor mask $p_i$ and bit $b_i = 1 \ll i$, define:
\begin{align*}
  \mathrm{input}_i  &= p_i \\
  \mathrm{output}_i &= p_i \cup b_i = p_i \;|\; b_i
\end{align*}
Under standard Petri firing with this mapping:
\[
  M' = (M \setminus p_i) \cup (p_i \cup b_i) = M \cup b_i
\]
\end{definition}

\begin{theorem}[Monotone done-set under mapped firing]
\label{thm:monotone_done}
Let $(M_0, M_1, \ldots)$ be the sequence of markings produced by
\texttt{PriorityPetriEngine::step()} with the mapping of Definition~\ref{def:monotone_map}.
Then $M_t \subseteq M_{t+1}$ for all $t \geq 0$.
\end{theorem}
\begin{proof}
Each \texttt{step()} fires a set $F_t \subseteq \text{enabled ops}$ of transitions.
For each $i \in F_t$: $M_{t+1} = M_t \cup b_i$.
Therefore $M_t \subseteq M_{t+1}$ by construction. By induction over $t$. \qed
\end{proof}

\subsection{Convergence}

\begin{corollary}[Termination]
On a finite $n$-op tape, the scheduler converges in at most $n$ ticks.
\end{corollary}
\begin{proof}
$M_t \in \B^n$ and $(M_t)_t$ is strictly increasing until $M_T = \top$ (all ops done).
Since $|\B^n| = 2^n$ is finite and $M_t \subsetneq M_{t+1}$ whenever progress is made,
the chain reaches $\top$ in at most $n$ steps. \qed
\end{proof}

\section{Branchless Scheduling: The $\mathcal{B}$-Calculus}
\label{sec:bcalculus}
%% ════════════════════════════════════════════════════════════════════════════

\subsection{Definition}

\begin{definition}[$\mathcal{B}$-Calculus]
The \emph{branchless calculus} $\mathcal{B}$ is the algebra over $\B^{64}$ generated by:
\begin{align*}
  \text{AND: }   & a \;\&\; b      & \text{NAND: } & \sim\!(a \;\&\; b)    \\
  \text{OR: }    & a \;|\; b       & \text{NOR: }  & \sim\!(a \;|\; b)     \\
  \text{XOR: }   & a \;\hat{}\; b  & \text{NOT: }  & \sim a                \\
  \text{SHL: }   & a \ll k         & \text{SHR: }  & a \gg k               \\
  \text{ADD: }   & a + b \pmod{2^{64}} & \text{SUB: } & a - b \pmod{2^{64}}
\end{align*}
Subject to: no conditional instructions (\texttt{if}/\texttt{match}) in the hot path.
\end{definition}

\paragraph{Why no branches?}
On a modern out-of-order processor, a mispredicted branch costs 10--20 cycles
(approximately 3--6\,ns at 3\,GHz).
At 2.07\,ns per scheduler tick, even one mispredicted branch doubles the tick cost.
The $\mathcal{B}$-Calculus eliminates this entirely.

\subsection{The SWAR Scheduler in $\mathcal{B}$}

The full scheduler tick can be written as a $\mathcal{B}$-expression:

\begin{align}
  \mathrm{candidates} &= \mathrm{check} \;\&\; \sim\mathrm{done}
  \label{eq:candidates}\\
  u_i        &= \mathrm{pred}_i \;\&\; \sim\mathrm{done}
  \label{eq:unmet}\\
  \mathrm{sat}_i      &= \overline{u_i \neq 0} = \texttt{wrapping\_neg}((u_i = 0)\;\text{as}\;\texttt{u64})
  \label{eq:sat}\\
  \mathrm{fire}_i     &= \mathrm{sat}_i \;\&\; (1 \ll i)
  \label{eq:fire}\\
  \mathrm{done}'      &= \mathrm{done} \;|\; \bigcup_i \mathrm{fire}_i
  \label{eq:done}\\
  \mathrm{check}'     &= \bigcup_{i:\mathrm{fire}_i \neq 0} \mathrm{succ}_i \;\&\; \sim\mathrm{done}'
  \label{eq:check}
\end{align}

Each equation uses only $\mathcal{B}$-operations.
The ``loop'' over $i$ in candidates is the \texttt{trailing\_zeros} bit-scan, which
iterates exactly as many times as there are set bits --- no wasted iterations.

\begin{proposition}[Bit-scan complexity]
The bit-scan loop \lstinline|while bits != 0 { i = bits.trailing_zeros(); bits &= bits - 1; }|
iterates exactly $|\{i : \mathrm{candidates}_i = 1\}|$ times.
Each iteration uses one \texttt{BSF}/\texttt{CTZ} instruction (1 cycle) and one
\texttt{BLSR} (1 cycle). Total: $2k$ cycles for $k$ candidates.
\end{proposition}

\subsection{TimeWheel: O(1) Deadline Algebra}

\begin{definition}[TimeWheel]
A TimeWheel of size $N$ is a pair $(S, t)$ where:
$S : \Z_N \to \B^{64}$ is the slot array, and $t \in \Z_N$ is the current pointer.

\textbf{schedule}$(\delta, b)$: $S[(t + \delta) \bmod N] \;\mathrel{|}=\; 1 \ll b$

\textbf{tick}: $\mathrm{due} := S[t]$;\; $S[t] := 0$;\; $t := (t+1) \bmod N$;\; \textbf{return} $\mathrm{due}$
\end{definition}

\begin{proposition}[O(1) tick]
Both \textbf{schedule} and \textbf{tick} are $O(1)$: one memory write and one
pointer increment respectively, with no iteration over scheduled events.
\end{proposition}

The key insight is that the \emph{slot} holds a bitmask of all events due at that
offset: multiple events at the same relative time cost the same as one.
This gives SLA deadline detection at \textbf{922\,ps} regardless of how many ops
have active SLA windows.

\subsection{LockFreeMpmcRing: Wait-Free CAS Dispatch}

\begin{definition}[Wait-free MPMC ring]
A \textsc{LockFreeMpmcRing} of capacity $N$ is a fixed-size array of $N$ CAS-protected
slots. \textbf{push\_t1} performs:
\[
  \texttt{CAS}(\&\mathrm{slot}[\mathrm{tail} \bmod N],\; \mathrm{empty},\; \mathrm{value})
\]
and returns immediately (non-blocking: \texttt{u32::MAX} on success, \texttt{0} on full ring).
\textbf{pop\_t1} is symmetric.
\end{definition}

\paragraph{Why CAS is sufficient.}
At any moment, at most one writer wins the CAS for a given slot.
Since $N=64$ and the POWL tape has at most 64 ops, the ring never overflows
during a single scheduler tick.
The wait-free property (no spinning, no backoff) is essential for the WCET guarantee.

\subsection{WcetFiber: Suspension Geometry}

The scheduler tick without fiber suspension serialises I/O latency:
\[
  T_{\text{serial}} = T_{\text{compute}} + T_{\text{IO}}
\]
With fiber suspension, I/O and compute overlap:
\[
  T_{\text{overlap}} = \max(T_{\text{compute}},\; T_{\text{IO}})
\]
The gain is $\Delta T = \min(T_{\text{compute}}, T_{\text{IO}})$.
For $T_{\text{IO}} \gg T_{\text{compute}}$ (the common case): $\Delta T \approx T_{\text{compute}}$,
recovering nearly all compute time at the cost of a 2.64\,ns fiber lifecycle.

\subsection{PriorityPetriEngine: Sub-Nanosecond Step}

\begin{proposition}[PetriEngine cost decomposition]
\texttt{PriorityPetriEngine<1,T>::step()} costs:
\[
  C_{\mathrm{step}}(T) = C_{\mathrm{init}} + T \cdot C_{\mathrm{fire}}
\]
where $C_{\mathrm{init}} \approx 5\,\text{ns}$ (stack allocation of $2T$ KBitSets)
and $C_{\mathrm{fire}} \approx 1\,\text{ns}$ per transition.
\end{proposition}

\begin{proof}[Measurement]
From benchmarks: $C_{\mathrm{step}}(8) = 581\,\text{ps}$ (dominated by $C_{\mathrm{fire}}$
when $C_{\mathrm{init}}$ is amortised by the pre-built engine).
$C_{\mathrm{step}}(64) = 102.7\,\text{ns}$, giving $C_{\mathrm{fire}} \approx 1.6\,\text{ns}$
per transition. The isolated engine cost is much lower than \texttt{petri\_tick} because
the latter re-constructs the engine each call. \qed
\end{proof}

\subsection{BLAKE3 Chain Homomorphism}

\begin{definition}[BLAKE3 receipt chain]
Let $\varepsilon \in \{0,1\}^{256}$ be the genesis hash (BLAKE3 of 32 zero bytes).
Define $h : \mathrm{Frame}^* \to \{0,1\}^{256}$ inductively:
\begin{align*}
  h(\varepsilon)        &= \mathrm{BLAKE3}(0^{32}) \\
  h(\sigma \cdot f)     &= \mathrm{BLAKE3}(h(\sigma) \,\|\, \mathrm{bytes}(f))
\end{align*}
\end{definition}

\begin{proposition}[Chain injectivity]
$h(\sigma) = h(\tau) \implies \sigma = \tau$ (with overwhelming probability $1 - 2^{-128}$
under the random oracle model for BLAKE3).
\end{proposition}

This means two distinct receipt chains \emph{cannot} share a hash.
The hash uniquely identifies the entire history of manufacture.
This is the cryptographic proof of $\Agency$: given $h(\sigma)$, no other sequence
of manufacturing steps could have produced it.

\subsection{ConformancePredicate: Q16.16 Fixed-Point Gate}

\begin{definition}[Q16.16 fixed-point]
A Q16.16 value $v \in \Z$ represents the rational $v / 2^{16}$.
The range $[0, 1]$ maps to $[0, 65536]$.
Comparison $a \geq b$ in Q16.16 is equivalent to integer $a \geq b$: no floating point.
\end{definition}

\begin{definition}[ConformancePredicate]
Given thresholds $(\theta_f, \theta_p, \theta_g, \theta_s) \in [0,1]^4$ in Q16.16
and metrics $(F, P, G, S) \in [0,1]^4$:
\[
  \mathrm{check}(F,P,G,S) = (F \geq \theta_f) \;\&\; (P \geq \theta_p)
                             \;\&\; (G \geq \theta_g) \;\&\; (S \geq \theta_s)
\]
All four comparisons are branchless integer comparisons ($\mathcal{B}$-calculus).
Pass: \textbf{395.9\,ps}. Fail: \textbf{395.2\,ps}.
\end{definition}

The \textbf{symmetric cost} (395 ps for both pass and fail) is the defining property
of a branchless gate: there is no prediction advantage from knowing the answer in advance.

\section{Completeness and Soundness}
\label{sec:soundness}
%% ════════════════════════════════════════════════════════════════════════════

\begin{theorem}[Scheduler completeness]
If the POWL tape has no cycle (guaranteed by the Kahn cycle-detection post-pass in the
compiler), then the scheduler reaches $\mathrm{done} = \top$ in at most $n$ ticks.
\end{theorem}
\begin{proof}
By acyclicity, there exists a topological order $\pi$ on the $n$ ops.
On tick $t$, all ops at topological depth $t$ have all predecessors in done
(by the invariant: $\mathrm{check} = $ enabled frontier).
Therefore at least one op fires per tick, and done grows by at least one bit. \qed
\end{proof}

\begin{theorem}[Receipt soundness]
If $\mathrm{receipt.chain\_hash} = h_k$, then there exists a unique sequence of frames
$(f_1,\ldots,f_k)$ satisfying the POWL model such that
$h_k = \mathrm{BLAKE3}(\mathrm{BLAKE3}(\cdots) \,\|\, \mathrm{bytes}(f_k))$.
\end{theorem}
\begin{proof}
By the collision-resistance of BLAKE3 under the random oracle model (probability $1 - 2^{-128}$)
and the injectivity of the chain function $h$.
The receipt is the unique pre-image. \qed
\end{proof}

\begin{theorem}[Zero unsafe in hot path]
All operations in \texttt{scheduler.rs} and \texttt{scheduler\_wired.rs} are safe Rust
(\texttt{\#![forbid(unsafe\_code)]} enforced at the crate level).
\end{theorem}
\begin{proof}
By compilation: the attribute \texttt{\#![forbid(unsafe\_code)]} on the crate root causes
the Rust compiler to emit a hard error if any \texttt{unsafe} block is encountered
in the crate. The CI build enforces this with \texttt{RUSTFLAGS="-D warnings"}.
103 tests pass with no errors and no warnings. \qed
\end{proof}

\subsection{The Ontological Claim}

The Chatman Equation $\Reality \vdash \Agency = \mu(\Ostar)$ is not a design metaphor.
It is a mathematical claim:

\begin{enumerate}
  \item $\Reality$ (the genesis receipt) is the axiom: it cannot be derived from nothing.
        It is the \emph{commitment} that $\Ostar$ will be applied.
  \item $\Ostar$ is the operator: one tick of the scheduler, one OCEL frame, one BLAKE3 extension.
  \item $\Agency = \mu(\Ostar)$ is the terminal state: the receipt chain that is its own
        proof of manufacture. $\Ostar(\Agency) = \Agency$ means no more enabled ops.
  \item The 395\,ps conformance gate is the per-step verification that $\Ostar$ was applied
        lawfully. It runs faster than a DRAM access. It is free to call at every boundary.
\end{enumerate}

\emph{The enterprise is not built around bcinr-powl.
bcinr-powl is the enterprise: the receipt chain is the data structure,
the conformance gate is the integrity check, and the BLAKE3 hash is the identity.}
